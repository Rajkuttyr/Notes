class Solution {

int count=0;

public int findTargetSumWays(int[] nums, int target) {

BackTrack(nums,target,0,0);

return count;

}

public void BackTrack(int[] n,int s,int curr,int ind){

if(ind==n.length){

if(s==curr){

count++;

}

return;

}

  
  

BackTrack(n,s,curr+n[ind],ind+1);

BackTrack(n,s,curr-n[ind],ind+1);

}

}
