class Solution {

List<List<Integer>> result = new ArrayList<>();

int[]arr ={1,2,3,4,5,6,7,8,9};

int k;

int n;

public List<List<Integer>> combinationSum3(int k, int n) {

this.k=k;

this.n=n;

BackTrack(0,0,new ArrayList<>());

return result;

}

public void BackTrack(int ind , int curr,List<Integer> temp){

if(temp.size()==k){

if(curr==n){

result.add(new ArrayList<>(temp));

}

return ;

}

if(curr>n){

return ;

}

if(ind==arr.length){

return ;

}

temp.add(arr[ind]);

BackTrack(ind+1,curr+arr[ind],temp);

temp.remove(temp.size()-1);

BackTrack(ind+1,curr,temp);

  

}

}
